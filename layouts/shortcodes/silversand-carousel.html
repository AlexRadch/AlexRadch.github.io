{{/* Карусель фотографий витрины Silver Sand. Первый параметр — ключ галереи в data/silversand.yaml, второй (необязательный) — подпись раздела для screen reader'а. Разметка одна на все разделы: свой счётчик, свои точки, свой кадр. */}} {{- $id := .Get 0 -}} {{- $label := .Get 1 | default "Фотографии" -}} {{- $slides := index hugo.Data.silversand $id -}} {{- if not $slides }}{{ errorf "Silver Sand: нет галереи %q в data/silversand.yaml" $id }}{{ end -}}
{{- $idx := 0 -}} {{- range $slide := $slides }} {{- $img := resources.Get (printf "silversand/%s" $slide.file) -}} {{- if not $img }}{{ errorf "Silver Sand: нет изображения %s" $slide.file }}{{ end -}} {{- $dims := "1600x" -}} {{- if lt $img.Width $img.Height }}{{ $dims = "x1600" }}{{ end -}} {{- $jpg := $img.Resize (printf "%s jpg q80" $dims) -}} {{- $webp := $img.Resize (printf "%s webp q80" $dims) -}} {{- $idx = add $idx 1 -}} {{- end }}
1 / {{ len $slides }}
